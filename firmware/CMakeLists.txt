cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(firmware C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

add_executable(firmware src/diag.c src/test_motor_single.c)
pico_generate_pio_header(firmware ${CMAKE_CURRENT_LIST_DIR}/quadrature_encoder.pio)
target_include_directories(firmware PRIVATE $(CMAKE_CURRENT_LIST_DIR)/inc)
target_link_libraries(firmware pico_stdlib hardware_pwm hardware_adc hardware_pio)
pico_enable_stdio_usb(firmware 0)
pico_enable_stdio_uart(firmware 1)
pico_add_extra_outputs(firmware)
